# 1. Cria a pasta do ambiente virtual na sua rota pessoal
python3 -m venv ~/venv

# 2. Atualiza o gerenciador de pacotes lá dentro
~/venv/bin/pip install --upgrade pip

# 3. Instala todas as bibliotecas que o bot precisa de uma vez só
~/venv/bin/pip install nerimity-sdk edge-tts pygame sounddevice
